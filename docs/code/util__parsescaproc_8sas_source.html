dotMySASProgram
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

89/**
90 @brief Collapses the per-line flow records into one record per grouping key.
91 @details Internal helper of %util_parseScaproc (uses l_stepFmt from the caller).
92 @param [in] p_inDsName= input flow data set
93 @param [out] p_outDsName= output data set
94 @param [in] p_byVarName= BY variable whose LAST. ends a group
95 @param [in] p_byVarNames= full BY statement variable list
96**/
97 %macro inner_pscp_getFlowDataSet(p_inDsName=
98 , p_outDsName=
99 , p_byVarName=
100 , p_byVarNames=);
101
102 /* Merge the data into one record for each step */
103 DATA &p_outDsName (KEEP=step step_name 'in'n 'out'n in_dsname out_dsname in_filename out_filename
104 in_entity_name out_entity_name unq_in_entity_name unq_out_entity_name);
105 LENGTH
106 step 5
107 step_name $60
108 in_dsname out_dsname $200
109 in_filename out_filename 'in'n 'out'n $200
110 in_entity_name out_entity_name unq_in_entity_name unq_out_entity_name $210;
111
112 RETAIN step step_name in_dsname out_dsname
113 in_filename out_filename
114 in_entity_name out_entity_name
115 unq_in_entity_name unq_out_entity_name;
116
117 DO UNTIL (last.&p_byVarName);
118 SET &p_inDsName(RENAME=(step=ds_step
119 dsname=ds_dsname filename=ds_filename entity_name=ds_entity_name));
120 BY &p_byVarNames;
121
122 if (first.&p_byVarName) then
123 CALL MISSING (step, step_name, in_dsname, out_dsname
124 , in_filename, out_filename, in_entity_name, out_entity_name
125 , unq_in_entity_name, unq_out_entity_name);
126
127 if (step = .) then step = ds_step;
128 if (step_name = '') then step_name = '_'||CATX("_",PUT(ds_step,&l_stepFmt.),PUT(ds_step,stepname.));
129 if (in_dsname = '') then if (strip(source)='i') then in_dsname = ds_dsname;
130 if (in_filename = '') then if (strip(source)='i') then in_filename = ds_filename;
131 if (in_entity_name = '') then if (strip(source)='i') then in_entity_name = ds_entity_name;
132 if (unq_in_entity_name = '') then if (strip(source)='i') then unq_in_entity_name = PUT(ds_entity_name,$unqEntity.);
133 if (out_dsname = '') then if (strip(source)='o') then out_dsname = ds_dsname;
134 if (out_filename = '') then if (strip(source)='o') then out_filename = ds_filename;
135 if (out_entity_name = '') then if (strip(source)='o') then out_entity_name = ds_entity_name;
136 if (unq_out_entity_name = '') then if (strip(source)='o') then unq_out_entity_name = PUT(ds_entity_name,$unqEntity.);
137 END;
138 'in'n = coalescec(in_dsname,in_filename);
139 'out'n = coalescec(out_dsname,out_filename);
140 if (CATS(in_dsname,out_dsname,in_filename,out_filename,in_entity_name,out_entity_name) ne '') then
141 OUTPUT;
142 RUN;
143 %mend inner_pscp_getFlowDataSet;

145/**
146 @brief Stage 1: reads the SCA text file; builds the concatenated-library format.
147 @details Internal helper of %util_parseScaproc. Shares the caller's %LOCAL macro variables
148 (l_*) and the WORK tables created by the preceding stages.
149**/
150%macro inner_pscp_readRecording;
151
152 /* Read the SCA text file */
153 DATA work.scadata(DROP=rc);
154 INFILE "&l_inScaprocFileName" LRECL=1000 LENGTH=linelength truncover END=eof;
155 LENGTH scaline $1000 file_linenum sca_linenum step 5 libref libname lineEnd $8;
156 LABEL
157 scaline = 'SCA line'
158 file_linenum = 'SCA file line number'
159 sca_linenum = 'SCA none empty line number'
160 step = 'Step Counter'
161 ;
162
163 if(_n_=1) then
164 do;
165 dcl hash libHash(ordered:'a');
166 libHash.definekey('libref');
167 libHash.definekey('libname');
168 libHash.definedone();
169 end;
170
171 INPUT @;
172 INPUT scaline $varying1000. linelength;
173 file_linenum+1;
174
175 if (length(strip(scaline)) gt 1) then
176 do;
177 sca_linenum+1;
178 if (_n_=1 OR index(scaline,'STEP SOURCE FOLLOWS')) then step+1 ;
179 /* Only JOBSPLIT lines are used downstream, so don't persist the rest */
180 if (index(scaline,'JOBSPLIT') > 0) then OUTPUT;
181
182 /* JOBSPLIT: CONCATMEM #C00003 SASHELP */
183 if (index(scaline,'CONCATMEM') gt 0) then
184 do;
185 scaline = SUBSTR(scaline, index(scaline,'CONCATMEM') + 10);
186 libref = SCAN(scaline,1,' ');
187 libname = SCAN(scaline,2,' ');
188 lineEnd = SCAN(scaline,3,' ');
189 if (libref ne '' and libname ne '' and lineEnd eq '*/') then rc = libHash.add();
190 end; /* End - (index(scaline,'CONCATMEM') gt 0) */
191 end; /* End - (length(strip(scaline)) gt 1) */
192
193 /* Export/write the hash table */
194 if (eof) then
195 do;
196 libHash.output(dataset: 'work.sca_libs');
197 end;
198 RUN;
199
200 /* Find the Max Step count */
201 PROC SQL NOPRINT;
202 SELECT CATS('Z',max(2,floor(log10(MAX(step,1)))+1),'.')
203 INTO :l_stepFmt
204 FROM work.scadata;
205 QUIT;
206 %if (&l_debug = Y) %then %put &=l_stepFmt;
207
208 /**
209 * Find Concatenated Libraries
210 */
211 DATA _null_;
212 if (0) then SET work.sca_libs nobs=nobs;
213 call symputx("l_concatLibObsCount",nobs);
214 RUN;
215
216 %if (&l_debug = Y) %then %put &=l_concatLibObsCount;
217 %if (&l_concatLibObsCount GT 0) %then
218 %do;
219 DATA work.sca_libs;
220 SET work.sca_libs(RENAME=(libref=start libname=label)) ;
221 RETAIN fmtname '$cncatlibs' type 'C';
222 RUN;
223 PROC FORMAT cntlin=work.sca_libs; RUN;
224 %end;
225 PROC DELETE DATA=work.sca_libs; RUN;
226
227
228%mend inner_pscp_readRecording;

230/**
231 @brief Stage 2: extracts step inputs/outputs/names/run-times and the stepname. format.
232 @details Internal helper of %util_parseScaproc. Shares the caller's %LOCAL macro variables
233 (l_*) and the WORK tables created by the preceding stages.
234**/
235%macro inner_pscp_extractIO;
236
237 /* Identify Input & Output data sets and their variables */
238 DATA work.scadata1 (KEEP=scaline sca_linenum step )
239 work.step_inputs (KEEP=step sca_linenum dsname variable type filename entity_name)
240 work.step_outputs (KEEP=step sca_linenum dsname variable type filename entity_name)
241 /* Always created: the stepname. format is needed even when p_includeSteps_yn=N */
242 work.step_runtimes (KEEP=step step_start_dt )
243 work.step_names(KEEP=step step_name type fmtname)
244 ;
245
246 /* Ignoring SASTemp outputs, Global Macro Variables and UTILITY files */
247 SET work.scadata(DROP=file_linenum WHERE=((index(scaline,"JOBSPLIT") > 0)
248 and (indexw(scaline,"SYMBOL")=0)
249 and (indexw(scaline,"CATALOG")=0)
250 and (index(scaline,".UTILITY")=0)
251 and (index(scaline,'DATASET OUTPUT SEQ WORK.SORTTMP')=0)
252 ));
253
254 LENGTH
255 dsname $50
256 variable $32
257 type $15
258 step_start_dt 8
259 filename $200
260 step_name $60
261 entity_name $210;
262
263 FORMAT step_start_dt datetime22.2;
264 filename='';
265
266 OUTPUT work.scadata1;
267
268 /* Evaluate each keyword test once per line */
269 is_attr = indexw(scaline,"ATTR") > 0;
270 is_file = indexw(scaline,"FILE") > 0;
271 is_dset = indexw(scaline,"DATASET") > 0;
272
273 IF ( (indexw(scaline,"TASKSTARTTIME") > 0)
274 OR (indexw(scaline,"JOBENDTIME") > 0) ) THEN
275 DO;
276 step_start_dt = INPUT(SCAN(scaline,4,' '), datetime22.2);
277 OUTPUT work.step_runtimes;
278 END;
279 ELSE IF(indexw(scaline,"PROCNAME") > 0) THEN
280 DO;
281 step_name = SCAN(scaline,4,' ');
282 type = 'N';
283 fmtname = 'stepName';
284 OUTPUT work.step_names;
285 END;
286 ELSE IF (is_attr OR is_dset OR is_file) THEN
287 DO;
288 IF (is_attr) THEN
289 DO;
290 scaline = SUBSTR(scaline,14);
291 dsname = SCAN(scaline,2,' ');
292 dsname = CATX('.',SCAN(dsname,1,'.'),SCAN(dsname,2,'.'));
293 %if (&l_concatLibObsCount GT 0) %then
294 %do;
295 dsname = CATX('.',PUT(SCAN(dsname,1,'.'),$cncatlibs.),SCAN(dsname,2,'.'));
296 %end;
297 in_flag = indexw(scaline,"INPUT");
298
299 /* Guard: skip malformed ATTR lines instead of calling SUBSTR with invalid positions */
300 if ( index(scaline,'VARIABLE:') = 0
301 OR index(scaline,'TYPE:') <= index(scaline,'VARIABLE:')
302 OR index(scaline,'LENGTH:') <= index(scaline,'TYPE:')
303 OR index(scaline,'LABEL:') <= index(scaline,'LENGTH:') ) then
304 do;
305 putlog 'NOTE: util_parseScaproc: skipped malformed ATTR line ' sca_linenum=;
306 DELETE;
307 end;
308 scaline = SUBSTR(scaline,index(scaline,'VARIABLE:'));
309 typ_idx = index(scaline,'TYPE:');
310 variable = substr(scaline,(10),(typ_idx - 10));
311
312 scaline = SUBSTR(scaline,index(scaline,'TYPE:'));
313 len_idx = index(scaline,'LENGTH:');
314 type = substr(scaline,(6),(len_idx - 6));
315
316 scaline = SUBSTR(scaline,index(scaline,'LENGTH:'));
317 lbl_idx = index(scaline,'LABEL:');
318 len = substr(scaline,(8),(lbl_idx - 8));
319
320 type = CATX(' ', IFC(type='CHARACTER','char','num') ,CATS('(',len,')') );
321 END;
322 ELSE IF (is_file) THEN
323 DO;
324 scaline = SUBSTR(scaline,14);
325 filename = SCAN(scaline,3,' ');
326 in_flag = indexw(scaline,"INPUT");
327 END;
328 ELSE IF (is_dset) THEN
329 DO;
330 scaline = SUBSTR(scaline,14);
331 dsname = SCAN(scaline,4,' ');
332 dsname = CATX('.',SCAN(dsname,1,'.'),SCAN(dsname,2,'.'));
333 %if (&l_concatLibObsCount GT 0) %then
334 %do;
335 dsname = CATX('.',PUT(SCAN(dsname,1,'.'),$cncatlibs.),SCAN(dsname,2,'.'));
336 %end;
337 in_flag = indexw(scaline,"INPUT");
338 END;
339
340 entity_name = ifc(COALESCEC(dsname,filename)='','',
341 CATS(ifc(in_flag > 0,'i','o'),'_',PUT(step,&l_stepFmt.),'_',COALESCEC(dsname,filename)));
342
343 /* Write the record to the correct destination based on the in_flag value */
344 IF (in_flag > 0) then OUTPUT work.step_inputs;
345 ELSE OUTPUT work.step_outputs;
346 END;
347 RUN;

389 /** -------

392 PROC SQL;
393 /* Unique Input/Output Data Sets */
394 CREATE TABLE work.unqEntity_fmtds AS
395 SELECT DISTINCT
396 entity_name as start length=210
397 ,CAT(STRIP(dsname),'_(',PUT(col_cnt,z4.),')') as label length=210
398 ,'$unqEntity' as fmtname
399 ,'C' as type

406 UNION
407 SELECT DISTINCT step,entity_name,dsname, count(*) as col_cnt
408 FROM work.step_outputs
409 WHERE variable ne ''
410 AND filename NOT IN ('SEQ','MULTI')
411 GROUP BY entity_name
412 )
413 ORDER BY 1;

421 FROM work.in_out_info_flow

423 AND strip(dsname) = '';

463 PROC SQL NOPRINT;
464 create table work.with_known_out as
465 select distinct coalesce(dsname,filename) as out, step
466 from work.step_outputs
467 group by 1
468 having count(*) = 1
469 order by step;

476 and PUT(t1.step,stepname.) in ('IMPORT', 'EXPORT')

515 %inner_pscp_getFlowDataSet(p_inDsName=work.impExp_flow

520 PROC SQL;
521 /* Create consolidated impExp flow data set */
522 CREATE TABLE work.impExp_flow3 AS
523 SELECT a.step
524 , a.step_name
525 , CASE
526 WHEN INDEX(a.step_name,'IMPORT') THEN coalesce(a.in_filename,b.in_filename)
527 ELSE coalesce(b.in_dsname,a.in_dsname)
528 END as 'in'n
529 , CASE
530 WHEN INDEX(a.step_name,'EXPORT') THEN coalesce(b.out_filename,a.out_filename)
531 ELSE coalesce(b.out_dsname,a.out_dsname)
532 END as 'out'n
533 , coalesce(b.in_entity_name,a.in_entity_name) as in_entity_name
534 , coalesce(b.out_entity_name,a.out_entity_name) as out_entity_name
535 , PUT(coalesce(b.in_entity_name,a.in_entity_name),$unqEntity.) as unq_in_entity_name length=210
536 , PUT(coalesce(b.out_entity_name,a.out_entity_name),$unqEntity.) as unq_out_entity_name length=210
537
538 FROM work.impExp_flow2(WHERE=(step IN (&l_impExpSteps))) AS a
539 FULL join work.impExp_flow2(WHERE=(step IN (&l_dataSteps))) AS b
540 on a.step = (b.step -1);
541
542 /* Exclude/Remove the import/Export steps from other data sets */
543 DELETE FROM work.step_runtimes WHERE step IN (&l_dataSteps);
544 DELETE FROM work.steps_in_out_flow WHERE step in (&l_impExpStepsExclude);
545 QUIT;
546
547 /* Add reformed Import/Export steps to the flow */
548 PROC APPEND BASE=work.steps_in_out_flow DATA=work.impExp_flow3 FORCE NOWARN; RUN;
549

551 PROC DATASETS LIB=WORK NOLIST;
552 DELETE impExp_flow:;
553 RUN; QUIT;
554 %end; /* End - (%superq(l_impExpSteps) NE ) */

556 /* Clean-up */
557 PROC DATASETS LIB=WORK NOLIST;
558 DELETE scadata: step_inputs step_outputs;
559 QUIT; RUN;
